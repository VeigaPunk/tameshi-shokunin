```python
async def search_all(key, pattern, path):
    chunks = []; skip = 0
    while True:
        got = await tool.grep(dict(i='Counting scoped search matches',
            pattern=pattern, path=path, case=False, gitignore=False, skip=skip))
        text = got['text']; chunks.append(text)
        m = re.search(r'Showing files (\d+)-(\d+) of (\d+)', text)
        if not m or int(m[2]) >= int(m[3]): break
        skip = int(m[2])
    text = '\n'.join(chunks)
    count = len(re.findall(r'^\*\d+:', text, re.M))
    files = len(re.findall(r'^#{2,} [^#\n]+#[0-9A-F]{4}', text, re.M))
    results[key] = (pattern, path, text, count, files)
    print(f'{key}: matched_lines={count} matched_files={files}')
```
