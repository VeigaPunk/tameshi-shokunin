#!/usr/bin/env python3
"""T87E1 counterfeit c: population read from the fixture worktree, sorted rows.

Receipts are found by os.walk over --repo/--root (the checked-out worktree),
not from committed blobs. Candidate states come from the candidate tree's
regular blobs via git, compared by sha256. Contract violation: worktree read.
"""
import argparse
import hashlib
import os
import re
import subprocess
import sys

ENV = {k: v for k, v in os.environ.items() if not k.startswith('GIT_')}
ENV.update(GIT_CONFIG_GLOBAL='/dev/null', GIT_CONFIG_NOSYSTEM='1',
           GIT_NO_REPLACE_OBJECTS='1')
REG = ('100644', '100755')
STRICT = {
    'base': r'^# design_base:[ \t]*([0-9a-f]{7,40})[ \t]*$',
    'tool': r'^# design_tool:[ \t]*(\S+)[ \t]*$',
    'sha': r'^# design_tool_sha256:[ \t]*([0-9a-f]{64})[ \t]*$',
    'marker': r'^# measured_bytes:[ \t]*worktree \(uncommitted at design_base\)[ \t]*$',
}
ANY_DESIGN = r'^# (?:measured_bytes|design_base|design_tool|design_tool_sha256):'
ANY_NORMAL = r'^# (?:measured_rev|measured_bytes_rev|tool|tool_sha256):'


def refuse(why):
    sys.stderr.write('design-blob-audit: setup refusal %s\n' % why)
    sys.exit(2)


def git(repo, *args):
    return subprocess.run(['git', '-C', repo, *args], capture_output=True, env=ENV)


def resolve(repo, rev):
    r = git(repo, 'rev-parse', '--verify', '--quiet', rev + '^{commit}')
    if r.returncode:
        refuse('unresolvable:' + rev)
    return r.stdout.decode().strip()


def tree(repo, rev):
    r = git(repo, 'ls-tree', '-r', '-z', rev)
    if r.returncode:
        refuse('tree-unreadable')
    out = {}
    for raw in r.stdout.split(b'\0'):
        if raw:
            meta, _, name = raw.partition(b'\t')
            mode, typ, oid = meta.decode().split()
            out[name.decode('utf-8', 'replace')] = (mode, typ, oid)
    return out


def parse(path, text):
    if re.search(ANY_DESIGN, text, re.M) is None:
        return None
    if re.search(ANY_NORMAL, text, re.M):
        refuse('mixed:' + path)
    for name in ('measured_bytes', 'design_base', 'design_tool', 'design_tool_sha256'):
        if len(re.findall(r'^# %s:' % name, text, re.M)) != 1:
            refuse('malformed:' + path)
    got = {k: re.findall(rx, text, re.M) for k, rx in STRICT.items()}
    if any(len(v) != 1 for v in got.values()):
        refuse('incomplete:' + path)
    return got['tool'][0], got['sha'][0]


def population_worktree(repo, root):
    recs = {}
    for dp, _, files in os.walk(os.path.join(repo, root)):
        for f in files:
            full = os.path.join(dp, f)
            if os.path.islink(full):
                continue
            rel = os.path.relpath(full, repo).replace(os.sep, '/')
            with open(full, 'rb') as fh:
                text = fh.read().decode('utf-8', 'replace').replace('\r\n', '\n')
            block = parse(rel, text)
            if block:
                recs[rel] = block
    if not recs:
        refuse('zero-population')
    return recs


def main():
    p = argparse.ArgumentParser()
    for f in ('--repo', '--receipt-rev', '--root', '--candidate-rev'):
        p.add_argument(f, required=True)
    a = p.parse_args()
    if a.root.startswith('/') or '..' in a.root.split('/'):
        refuse('root:' + a.root)
    resolve(a.repo, a.receipt_rev)
    cr = resolve(a.repo, a.candidate_rev)
    recs = population_worktree(a.repo, a.root)
    ct = tree(a.repo, cr)
    rows, carried = [], 0
    for path in sorted(recs):
        tool, sha = recs[path]
        ent = ct.get(tool)
        if ent is None:
            state = 'absent'
        elif ent[0] not in REG or ent[1] != 'blob':
            state = 'nonregular'
        else:
            data = git(a.repo, 'cat-file', 'blob', ent[2]).stdout
            state = 'carried' if hashlib.sha256(data).hexdigest() == sha else 'different'
        carried += state == 'carried'
        rows.append('DESIGN_BLOB_ROW %s state=%s' % (path, state))
    rows.append('DESIGN_BLOB: receipts=%d carried=%d orphan=%d'
                % (len(recs), carried, len(recs) - carried))
    sys.stdout.write('\n'.join(rows) + '\n')
    return 0 if carried == len(recs) else 1


if __name__ == '__main__':
    sys.exit(main())
