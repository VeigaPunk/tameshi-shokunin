`# measured_rev:`, `# tool:` and `# tool_sha256:` header line.
States: bound / corrected / unbound (named reason).
Exit 0 iff unbound == 0; exit 1 if any row is unbound; setup exit 2 before any
row effect on a zero population, a duplicate/malformed header, a non-repo
--repo, or a non-repo-relative tool path.
"""
import argparse
import hashlib
import os
import posixpath
import re
import shutil
import subprocess
import sys
import tempfile

MARK_MEASURED = b"# measured_rev: "
MARK_TOOL = b"# tool: "
MARK_SHA = b"# tool_sha256: "
MARK_CORR = b"# measured_bytes_rev: "

SHA_RE = re.compile(r"^[0-9a-fA-F]{64}$")
REV_RE = re.compile(r"^[0-9a-fA-F]{7,40}$")


class SetupRefusal(Exception):
    def __init__(self, reason):
        super().__init__(reason)
        self.reason = reason


def _git(repo, *args):
    p = subprocess.run(["git", "-C", repo, *args], capture_output=True)
    return p.returncode, p.stdout


def _sha256_file(path):
    """Return (kind, value): kind in {sha, absent-marker...} not needed; this
    hashes the file bytes and parses headers. Raises SetupRefusal."""
    with open(path, "rb") as f:
        raw = f.read()
    measured, tool, sha, corr = [], [], [], []
    corr_line_no = -1
    sha_line_no = -1
    for i, ln in enumerate(raw.split(b"\n")):
        s = ln.rstrip(b"\r")
        hit = None
        for key, mark in (
            ("measured", MARK_MEASURED),
            ("sha", MARK_SHA),
            ("corr", MARK_CORR),
            ("tool", MARK_TOOL),
        ):
            if s.startswith(mark):
                hit = (key, s[len(mark):].strip())
                break
        if hit is None:
            continue
        key, val = hit
        if not val:
            raise SetupRefusal("malformed-header")
        if key == "measured":
            measured.append(val)
        elif key == "sha":
            sha.append(val)
            sha_line_no = i
        elif key == "corr":
            corr.append(val)
            corr_line_no = i
        else:
            tool.append(val)
        if len(measured) > 1 or len(tool) > 1 or len(sha) > 1 or len(corr) > 1:
            raise SetupRefusal("duplicate-header")
    if not (measured and tool and sha):
        if measured or tool or sha or corr:
            raise SetupRefusal("malformed-header")
        return None  # not a receipt: carries no header markers
    m = measured[0].decode("ascii", "replace")
    t = tool[0].decode("utf-8", "replace")
    s = sha[0].decode("ascii", "replace")
    c = corr[0].decode("ascii", "replace") if corr else None
    if not REV_RE.match(m):
        raise SetupRefusal("malformed-header")
    if not SHA_RE.match(s):
        raise SetupRefusal("malformed-header")
    if c is not None and not REV_RE.match(c):
        raise SetupRefusal("malformed-header")
    if c is not None and (sha_line_no < 0 or corr_line_no < sha_line_no):
        raise SetupRefusal("correction-not-append-only")
    return {"measured": m, "tool": t, "sha": s.lower(), "corr": c}


def _check_tool_path(tool):
    if tool != tool.strip() or not tool:
        raise SetupRefusal("tool-path-escape")
    if os.path.isabs(tool) or tool.startswith("~"):
        raise SetupRefusal("tool-path-escape")
    norm = posixpath.normpath(tool.replace(os.sep, "/"))
    if norm in (".", "..") or norm.startswith("../"):
        raise SetupRefusal("tool-path-escape")
    return norm


def scan_population(root):
    if not os.path.isdir(root):
        raise SetupRefusal("root-missing")
    entries = []
    for dirpath, dirnames, filenames in os.walk(root):
        dirnames.sort()
        for name in sorted(filenames):
            p = os.path.join(dirpath, name)
            if not os.path.isfile(p):
                continue  # skips special files; regular files only
            rel = os.path.relpath(p, root).replace(os.sep, "/")
            base = root.replace(os.sep, "/").rstrip("/")
            disp = base + "/" + rel
            entry = _sha256_file(p)
            if entry is None:
                continue  # non-receipt: skipped, never a refusal
            entry["tool"] = _check_tool_path(entry["tool"])
            entry["display"] = disp
            entries.append(entry)
    return entries


def evaluate(repo, e):
    measured, tool, rec, corr = e["measured"], e["tool"], e["sha"], e["corr"]
    rc, _ = _git(repo, "rev-parse", "--verify", "--quiet", measured + "^{object}")
    blob = None
    if rc == 0:
        rc2, data = _git(repo, "cat-file", "blob", measured + ":" + tool)
        if rc2 == 0:
            blob = hashlib.sha256(data).hexdigest()
    if blob == rec:
        return "bound", "measured-rev-carries", measured[:12]
    if rc != 0:
        m_detail = "measured-rev-absent"
    elif blob is None:
        m_detail = "tool-absent-at-rev"
    else:
        m_detail = "bytes-rev-absent"
    if corr:
        crc, _ = _git(repo, "rev-parse", "--verify", "--quiet", corr + "^{object}")
        if crc != 0:
            return "unbound", "correction-rev-absent", "-"
        rc3, cdata = _git(repo, "cat-file", "blob", corr + ":" + tool)
        if rc3 == 0 and hashlib.sha256(cdata).hexdigest() == rec:
            return "corrected", "bytes-rev-carries", corr[:12]
        return "unbound", "bytes-rev-mismatch", "-"
    return "unbound", m_detail, "-"


def cmd(argv):
    """Run the CLI. Returns (exit_code, stdout_text)."""
    ap = argparse.ArgumentParser(prog="receipt-bytes-bind")
    ap.add_argument("--repo", default=".")
    ap.add_argument("--root", default=".")
    ap.add_argument("--selftest", action="store_true")
    args = ap.parse_args(argv)
    if args.selftest:
        return _selftest()
    out = []
    rc, _ = _git(args.repo, "rev-parse", "--show-toplevel")
    if rc != 0:
        return 2, "RECEIPT_BIND_SETUP: not-a-git-repo\n"
    try:
        entries = scan_population(args.root)
    except SetupRefusal as r:
        return 2, "RECEIPT_BIND_SETUP: %s\n" % r.reason
    if not entries:
        return 2, "RECEIPT_BIND_SETUP: zero-population\n"
    counts = {"bound": 0, "corrected": 0, "unbound": 0}
    for e in sorted(entries, key=lambda x: x["display"]):
        state, detail, carrying = evaluate(args.repo, e)
        counts[state] += 1
        out.append(
            "RECEIPT_BIND_ROW %s measured_rev=%s tool=%s recorded=%s carrying=%s "
            "state=%s detail=%s"
            % (
                e["display"],
                e["measured"][:12],
                e["tool"],
                e["sha"][:12],
                carrying,
                state,
                detail,
            )
        )
    out.append(
        "RECEIPT_BIND: receipts=%d bound=%d corrected=%d unbound=%d"
        % (len(entries), counts["bound"], counts["corrected"], counts["unbound"])
    )
    return (1 if counts["unbound"] else 0), "\n".join(out) + "\n"


# ---------------- selftest ----------------

def _mkrepo(d):
    os.makedirs(d, exist_ok=True)
    assert subprocess.run(
        ["git", "-C", d, "rev-parse", "--show-toplevel"], capture_output=True
    ).returncode != 0
    subprocess.run(["git", "-C", d, "init", "-q"], check=True)
    top = subprocess.run(
        ["git", "-C", d, "rev-parse", "--show-toplevel"],
        capture_output=True,
    ).stdout.decode().strip()
    assert top == os.path.realpath(d), top
    ident = ["-c", "user.name=t", "-c", "user.email=t@example"]
    subprocess.run(["git", "-C", d, *ident, "commit", "--allow-empty", "-q", "-m", "c0"], check=True)
    return ident


def _commit(repo, ident, path, data, msg):
    full = os.path.join(repo, path)
    os.makedirs(os.path.dirname(full), exist_ok=True)
    with open(full, "wb") as f:
        f.write(data)
    subprocess.run(["git", "-C", repo, "add", path], check=True)
    subprocess.run(["git", "-C", repo, *ident, "commit", "-q", "-m", msg], check=True)
    return subprocess.run(
        ["git", "-C", repo, "rev-parse", "HEAD"], capture_output=True
    ).stdout.decode().strip()


def _receipt(dirpath, name, measured, tool, sha, corr=None, extra_measured=None):
    os.makedirs(dirpath, exist_ok=True)
    lines = [
        "# receipt: %s" % name,
        "# measured_rev: %s" % measured,
    ]
    if extra_measured:
        lines.append("# measured_rev: %s" % extra_measured)
    lines += ["# tool: %s" % tool, "# tool_sha256: %s" % sha]
    if corr:
        lines.append("# measured_bytes_rev: %s" % corr)
    p = os.path.join(dirpath, name)
    with open(p, "w") as f:
        f.write("\n".join(lines) + "\n")
    return p


def _selftest():
    arms = []
    tmp = tempfile.mkdtemp(prefix="rbb-selftest-")
    try:
        X, Y, Z = b"TOOL-X\n", b"TOOL-Y\n", b"TOOL-NEVER\n"
        shaX = hashlib.sha256(X).hexdigest()
        shaY = hashlib.sha256(Y).hexdigest()
        shaZ = hashlib.sha256(Z).hexdigest()

        def fixture(name):
            repo = os.path.join(tmp, name)
            ident = _mkrepo(repo)
            a = _commit(repo, ident, "t.txt", X, "x")
            b = _commit(repo, ident, "t.txt", Y, "y")
            root = os.path.join(tmp, name + "-root")
            os.makedirs(root, exist_ok=True)
            return repo, root, a, b

        def expect(name, cond_ok, detail):
            arms.append((name, cond_ok, detail))

        # ARM1 bound
        repo, root, a, b = fixture("arm1")
        _receipt(root, "r.txt", a, "t.txt", shaX)
        rc, out = cmd(["--repo", repo, "--root", root])
        expect("bound", rc == 0 and "state=bound" in out and "detail=measured-rev-carries" in out
               and "RECEIPT_BIND: receipts=1 bound=1 corrected=0 unbound=0" in out,
               "rc=%d out=%s" % (rc, out))

        # ARM2 unbound named
        repo, root, a, b = fixture("arm2")
        _receipt(root, "r.txt", a, "t.txt", shaZ)
        rc, out = cmd(["--repo", repo, "--root", root])
        expect("unbound_named", rc == 1 and "state=unbound" in out
               and "detail=bytes-rev-absent" in out
               and "RECEIPT_BIND: receipts=1 bound=0 corrected=0 unbound=1" in out,
               "rc=%d out=%s" % (rc, out))

        # ARM3 corrected
        repo, root, a, b = fixture("arm3")
        _receipt(root, "r.txt", a, "t.txt", shaY, corr=b)
        rc, out = cmd(["--repo", repo, "--root", root])
        expect("corrected", rc == 0 and "state=corrected" in out
               and "detail=bytes-rev-carries" in out
               and "RECEIPT_BIND: receipts=1 bound=0 corrected=1 unbound=0" in out,
               "rc=%d out=%s" % (rc, out))

        # ARM4 false correction refused
        repo, root, a, b = fixture("arm4")
        _receipt(root, "r.txt", a, "t.txt", shaZ, corr=b)
        rc, out = cmd(["--repo", repo, "--root", root])
        expect("false_correction_refused", rc == 1 and "state=unbound" in out
               and "detail=bytes-rev-mismatch" in out and "carrying=-" in out,
               "rc=%d out=%s" % (rc, out))

        # ARM5 unresolvable correction refused
        repo, root, a, b = fixture("arm5")
        _receipt(root, "r.txt", a, "t.txt", shaZ, corr="deadbeef000")
        rc, out = cmd(["--repo", repo, "--root", root])
        expect("unresolvable_correction_refused", rc == 1
               and "detail=correction-rev-absent" in out,
               "rc=%d out=%s" % (rc, out))

        # ARM6 duplicate header refused
        repo, root, a, b = fixture("arm6")
        _receipt(root, "r.txt", a, "t.txt", shaX, extra_measured=b)
        rc, out = cmd(["--repo", repo, "--root", root])
        expect("duplicate_header_refused", rc == 2 and "RECEIPT_BIND_SETUP:" in out
               and "RECEIPT_BIND_ROW" not in out,
               "rc=%d out=%s" % (rc, out))

        # ARM7 zero population refused
        repo, root, a, b = fixture("arm7")
        with open(os.path.join(root, "note.txt"), "w") as f:
            f.write("no markers here\n")
        rc, out = cmd(["--repo", repo, "--root", root])
        expect("zero_population_refused", rc == 2 and "zero-population" in out,
               "rc=%d out=%s" % (rc, out))

        # ARM8 tool absent -> unbound
        repo, root, a, b = fixture("arm8")
        _receipt(root, "r.txt", a, "missing.txt", shaX)
        rc, out = cmd(["--repo", repo, "--root", root])
        expect("tool_absent_unbound", rc == 1 and "detail=tool-absent-at-rev" in out,
               "rc=%d out=%s" % (rc, out))

        # ARM9 non-repo refused
        plain = os.path.join(tmp, "plain")
        os.makedirs(plain, exist_ok=True)
        rc, out = cmd(["--repo", plain, "--root", tmp])
        expect("non_repo_refused", rc == 2 and "not-a-git-repo" in out,
               "rc=%d out=%s" % (rc, out))

        # ARM10 tool path escape refused
        repo, root, a, b = fixture("arm10")
        _receipt(root, "r.txt", a, "../x.py", shaX)
        rc, out = cmd(["--repo", repo, "--root", root])
        expect("tool_path_escape_refused", rc == 2 and "tool-path-escape" in out
               and "RECEIPT_BIND_ROW" not in out,
               "rc=%d out=%s" % (rc, out))
    finally:
        shutil.rmtree(tmp, ignore_errors=True)
    npass = sum(1 for _, ok, _ in arms if ok)
    lines = []
    for i, (name, ok, detail) in enumerate(arms, 1):
        lines.append("SELFTEST_ARM%d_%s: %s" % (i, name, "PASS" if ok else "FAIL " + detail))
    lines.append("RECEIPT_BIND_SELFTEST: %d/%d %s"
                 % (npass, len(arms), "pass" if npass == len(arms) else "fail"))
    return (0 if npass == len(arms) else 1), "\n".join(lines) + "\n"


def main(argv=None):
    rc, out = cmd(sys.argv[1:] if argv is None else argv)
    sys.stdout.write(out)
    return rc


if __name__ == "__main__":
    sys.exit(main())
