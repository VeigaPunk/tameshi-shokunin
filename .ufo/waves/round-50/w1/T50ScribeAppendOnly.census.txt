def g(*a):

dirs=['.ufo/waves/round-%d'%n for n in range(43,51)]
raw=g('log','--reverse','--format=%x01%H','--name-only','--',*dirs).decode()
cur=None; hist=collections.defaultdict(list)
for line in raw.splitlines():
    if line.startswith('\x01'): cur=line[1:].strip()
    elif line.strip(): hist[line.strip()].append(cur)
cand=[f for f in g('ls-tree','-r','--name-only','HEAD','.ufo/waves/').decode().splitlines()

multi=[f for f in cand if len(hist.get(f,[]))>1]
rew=[]; pairs=0
for f in sorted(multi):
    cs=hist[f]
    for a,b in zip(cs,cs[1:]):
        pairs+=1
        ba=g('cat-file','blob','%s:%s'%(a,f)); bb=g('cat-file','blob','%s:%s'%(b,f))
        if not bb.startswith(ba):

            while n<m and ba[n]==bb[n]: n+=1
            rew.append((f,a[:8],b[:8],len(ba)-n,len(ba),len(bb)))
print('md seat files=%d multi_commit=%d pairs=%d rewrite_pairs=%d files_with_rewrite=%d'%
      (len(cand),len(multi),pairs,len(rew),len({r[0] for r in rew})))
for f,a,b,lost,o,n in rew:
    print('REWRITE %s %s->%s lost=%d old=%d new=%d'%(f,a,b,lost,o,n))
