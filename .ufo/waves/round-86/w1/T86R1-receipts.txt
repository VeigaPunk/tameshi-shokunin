import json, pathlib, re, subprocess, sys, tempfile
R = pathlib.Path('/home/vgpnk/Projects/shokunin/tameshi-shokunin')

REV = '55d5a5c6002a048337cc511e1712df87b961d364'
MEAS = R / '.ufo/evidence/round-86/measure.py'
def run(tool, tag):
    work = S / 'works' / tag
    work.mkdir(parents=True, exist_ok=True)
    for f in list(work.iterdir()):
        pass
    cmd = [sys.executable, '-I', str(MEAS), '--repo', str(R), '--rev', REV,
           '--work', str(work)]
    if tool:
        cmd += ['--tool', str(tool)]
    p = subprocess.run(cmd, capture_output=True, text=True, cwd=str(S))
    out = p.stdout
    final = out.strip().splitlines()[-1] if out.strip() else '(no stdout)'
    failing = []
    m = re.search(r'\{.*\}', out, re.S)
    if m:
        try:
            data = json.loads(m.group(0))
            failing = [r['case'] + '(exit=%s/want=%s,stdout_ok=%s)' % (
                r['exit'], r['expected_exit'], r['stdout'] == r['expected_stdout'].strip())
                for r in data['results'] if not r['ok']]
        except Exception as e:
            failing = ['json-parse-error:%s' % e]
    return p.returncode, final, failing
mode = sys.argv[1]
if mode == 'mut':
    tools = [('base(frozen tool)', S / 'mut/base.py')] + [
        (p.stem, p) for p in sorted((S / 'mut').glob('M*.py'))]
    for label, t in tools:
        rc, final, failing = run(t, 'm-' + label.split(' ')[0])
        print('%-36s rc=%d  %s' % (label, rc, final))
        for f in failing: print('      FAIL case:', f)
elif mode == 'seq':
    counts = []
    for i in range(int(sys.argv[2])):
        rc, final, failing = run(R / '.ufo/evidence/round-86/controls/ctl-sequence.py', 'seq-%d' % i)
        counts.append(final.split('passed=')[1].split()[0] if 'passed=' in final else final)
