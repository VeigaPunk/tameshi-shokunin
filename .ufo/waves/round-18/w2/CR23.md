`drift.py` (sha256 `99c4a5b6fb6e37501f1e40094e576d3b8dfd7c6f89466e74347d4fa0252acb47`):

```python
#!/usr/bin/env python3
"""Pointer-drift check for a SKILL.md line pointer, numbers only.

Prints the numeric hunk ranges of `git diff -U0 OLD NEW -- SKILL.md` and the
net line shift each pointer line accumulates; never prints diff content.
"""
import re
import subprocess
import sys

old, new, *ptrs = sys.argv[1:]
ptrs = [int(p) for p in ptrs]
out = subprocess.run(
    ["git", "diff", "--no-color", "--no-ext-diff", "-U0", old, new, "--", "SKILL.md"],
    check=True, capture_output=True,
).stdout.decode("utf-8", "replace")
hunk = re.compile(r"^@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@")
hunks = []
for line in out.splitlines():
    m = hunk.match(line)
    if m:
        os_, oc, ns, nc = m.groups()
        hunks.append((int(os_), 1 if oc is None else int(oc), int(ns), 1 if nc is None else int(nc)))
print(f"hunks={len(hunks)}")
for o_s, o_c, n_s, n_c in hunks:
    print(f"  old {o_s},{o_c} -> new {n_s},{n_c}  delta={n_c - o_c:+d}")
for p in ptrs:
    shift = 0
    touched = False
    for o_s, o_c, n_s, n_c in hunks:
        # with -U0, a pure insertion (o_c == 0) is placed *after* old line o_s
        if o_c == 0:
            if o_s < p:
                shift += n_c
        else:
            if o_s <= p <= o_s + o_c - 1:
                touched = True
            elif o_s + o_c - 1 < p:
                shift += n_c - o_c
    status = "line itself changed" if touched else f"maps to new line {p + shift}"
    print(f"pointer old:{p} -> shift {shift:+d}; {status}")
```

`sched_scan.py` (sha256 `e817e1a6f574149e3029e86c662aee557de871d2b30fa9265c239cca6ac8db57`):

```python
#!/usr/bin/env python3
"""Scan code files at a rev for scheduler/daemon primitives.

Skips every path the cold-read bounds forbid (LOG.md, SCORECARD.md, SKILL.md,
plans/, reports/, corpus/lessons/, .ufo/waves/) and every cold-read seat's
receipts (.ufo/evidence/*/coldread/); prints rev, scanned-file count and the
`git grep -n` matches.
"""
import re
import subprocess
import sys

FORBIDDEN_EXACT = {"LOG.md", "SCORECARD.md", "SKILL.md"}
FORBIDDEN_PREFIX = ("plans/", "reports/", "corpus/lessons/", ".ufo/waves/")
CODE = re.compile(r"(\.(sh|bash|py|mjs|cjs|js|ts|service|timer|socket|path|cron)|/Makefile|^Makefile|/justfile|^justfile)$")
PATTERN = (
    r"systemd-run|OnCalendar|OnUnitActiveSec|crontab|list-timers|\.timer\b|--on-calendar|"
    r"\bnohup\b|daemoni[sz]e|\bdaemon\b|\bat now\b|inotifywait|while true|while :;"
)

for rev in sys.argv[1:]:
    names = subprocess.run(["git", "ls-tree", "-r", "--name-only", rev],
                           check=True, capture_output=True, text=True).stdout.splitlines()
    keep = [n for n in names
            if n not in FORBIDDEN_EXACT
            and not n.startswith(FORBIDDEN_PREFIX)
            and "/coldread/" not in n
            and CODE.search(n)]
    print(f"rev={rev} tree_files={len(names)} scanned_code_files={len(keep)}")
    if not keep:
        continue
    res = subprocess.run(["git", "grep", "-n", "-I", "-E", PATTERN, rev, "--", *keep],
                         capture_output=True, text=True)
    hits = res.stdout.splitlines()
    print(f"  git_grep_exit={res.returncode} matches={len(hits)}")
    for h in hits:
        print("  " + h[:240])
```
