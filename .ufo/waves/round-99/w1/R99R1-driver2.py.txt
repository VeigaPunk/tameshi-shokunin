#!/usr/bin/env python3
import os, subprocess, sys, tempfile, shutil

LANE = "/home/vgpnk/Projects/shokunin/tameshi-shokunin"
TOOL = os.path.join(LANE, "tools/tally-reconcile.py")
GIT = "/usr/bin/git"
ENV = {"PATH": "/usr/local/bin:/usr/bin:/bin", "LANG": "C.UTF-8", "LC_ALL": "C.UTF-8",
       "GIT_AUTHOR_NAME": "fx", "GIT_AUTHOR_EMAIL": "fx@fixture.invalid",
       "GIT_COMMITTER_NAME": "fx", "GIT_COMMITTER_EMAIL": "fx@fixture.invalid"}

def run_case(label, files, record, roots=()):
    d = tempfile.mkdtemp(dir="/fast/scratch/hyaku/tameshi-r99/R99R1")
    def g(*a):
        return subprocess.run([GIT, "-C", d] + list(a), capture_output=True, env=ENV)
    assert g("init", "-q").returncode == 0
    top = g("rev-parse", "--show-toplevel").stdout.decode().strip()
    assert os.path.realpath(top) == os.path.realpath(d), "toplevel mismatch"
    for p, body in files.items():
        fp = os.path.join(d, p)
        os.makedirs(os.path.dirname(fp) or d, exist_ok=True)
        mode = "wb" if isinstance(body, bytes) else "w"
        with open(fp, mode) as fh:
            fh.write(body)
    assert g("add", "-A").returncode == 0
    assert g("commit", "-q", "-m", "fx").returncode == 0
    rev = g("rev-parse", "HEAD").stdout.decode().strip()
    cmd = ["python3", "-I", TOOL, "--repo", d, "--rev", rev, "--record", record]
    for r in roots:
        cmd += ["--receipt-root", r]
    p = subprocess.run(cmd, capture_output=True, text=True)
    summ = [l for l in p.stdout.splitlines() if l.startswith("TALLY_RECONCILE")]
    tall = [l for l in p.stdout.splitlines() if l.startswith("TALLY ")]
    print(f"### {label} rc={p.returncode}")
    for l in tall:
        print("   " + l)
    print("   SUMMARY: " + (" | ".join(summ) if summ else "(no summary)"))
    if p.stderr.strip():
        print("   STDERR: " + p.stderr.strip()[:200])
    shutil.rmtree(d, ignore_errors=True)

REC = "p/round-3.md"
EV = ".ufo/evidence/round-3/a.out"
# window recount: 'selftest' + exactly 12 chars then N/M; and 13
run_case("tally-window-12-exact2",
         {REC: "selftest " + "k" * 11 + "2/2 (`a.out`)\n", EV: "C_SELFTEST: 2/2 pass\n"}, REC)
run_case("tally-window-13-over2",
         {REC: "selftest " + "k" * 12 + "2/2 (`a.out`)\n", EV: "C_SELFTEST: 2/2 pass\n"}, REC)
# gap is exactly 12 chars after 'selftest': space + 11 k = 12
# citation: path with spaces — receipt file name with space
run_case("cite-path-with-space",
         {REC: "selftest 2/2 (`my file.out`)\n", ".ufo/evidence/round-3/my file.out": "C_SELFTEST: 2/2 pass\n"}, REC)
# citation: backtick-wrapped vs bare (no backticks)
run_case("cite-bare-no-backtick",
         {REC: "selftest 2/2 (a.out)\n", EV: "C_SELFTEST: 2/2 pass\n"}, REC)
# citation: .. segments
run_case("cite-dotdot",
         {REC: "selftest 2/2 (`../evidence/round-3/a.out`)\n", EV: "C_SELFTEST: 2/2 pass\n"}, REC)
run_case("cite-dotdot-resolve",
         {REC: "selftest 2/2 (`x/../../round-3/a.out`)\n", EV: "C_SELFTEST: 2/2 pass\n"}, REC)
# absolute path citation (leading /)
run_case("cite-absolute",
         {REC: "selftest 2/2 (`/ufo/evidence/round-3/a.out`)\n", EV: "C_SELFTEST: 2/2 pass\n"}, REC)
# path that is the record itself
run_case("cite-record-itself",
         {REC: "selftest 2/2 (`p/round-3.md`)\n", "p/round-3.md": "selftest 2/2 (`p/round-3.md`)\n"}, REC)
# URL with scheme
run_case("cite-url-https",
         {REC: "selftest 2/2 see https://h.example/x/a.out\n", EV: "C_SELFTEST: 2/2 pass\n"}, REC)
# URL without scheme: www.host/path
run_case("cite-url-www",
         {REC: "selftest 2/2 see www.h.example/x/a.out\n", EV: "C_SELFTEST: 2/2 pass\n"}, REC)
# host with dot but no scheme: h.example/x/a.out
run_case("cite-host-no-scheme",
         {REC: "selftest 2/2 see h.example/x/a.out\n", EV: "C_SELFTEST: 2/2 pass\n"}, REC)
# .json5 extension
run_case("cite-json5",
         {REC: "selftest 2/2 (`a.json5`)\n", ".ufo/evidence/round-3/a.json5": "C_SELFTEST: 2/2 pass\n"}, REC)
# .outfile extension
run_case("cite-outfile",
         {REC: "selftest 2/2 (`a.outfile`)\n", ".ufo/evidence/round-3/a.outfile": "C_SELFTEST: 2/2 pass\n"}, REC)
# .out2 extension ending digit after
run_case("cite-out2",
         {REC: "selftest 2/2 (`a.out2`)\n", ".ufo/evidence/round-3/a.out2": "C_SELFTEST: 2/2 pass\n"}, REC)
# extension at end of token with backtick before: fine. Citation followed by period?
run_case("cite-trailing-period",
         {REC: "selftest 2/2 (`a.out`).\n", EV: "C_SELFTEST: 2/2 pass\n"}, REC)
print("DONE-batch2")
