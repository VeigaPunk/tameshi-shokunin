import glob
import re

# Any textual N/M shape as it would appear inside a Python regex literal or
# format string: `[0-9]+/[0-9]+`, `\d+/\d+`, `(\d+)/(\d+)`.
N_M = re.compile(r"\[0-9\]\+/\[0-9\]\+|\\d\+/\\d\+|\(\\d\+\)/\(\\d\+\)")
CITE = re.compile(r"\.(out|txt|log|json)\b")

scanned = 0
both = []
tally_only = []
for p in sorted(glob.glob("*/tools/**/*.py", recursive=True)):
    if p.startswith("tameshi-shokunin"):
        continue
    try:
        s = open(p, encoding="utf-8", errors="replace").read()
    except OSError:
        continue
    scanned += 1
    has_nm = bool(N_M.search(s))
    has_cite = bool(CITE.search(s)) and "receipt" in s.lower()
    if has_nm and has_cite:
        both.append(p)
    elif has_nm:
        tally_only.append(p)

print("scanned sibling tools:", scanned)
print("parse an N/M tally shape:", len(both) + len(tally_only))
print("N/M tally shape AND receipt-path handling:", len(both))
for p in both:
    print("  both:", p)
print("N/M tally shape only (first 10):")
for p in tally_only[:10]:
    print("  tally:", p)
