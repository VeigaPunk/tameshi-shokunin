#!/usr/bin/env python3
import os, subprocess, tempfile, shutil

LANE = "/home/vgpnk/Projects/shokunin/tameshi-shokunin"
TOOL = os.path.join(LANE, "tools/tally-reconcile.py")
GIT = "/usr/bin/git"
ENV = {"PATH": "/usr/local/bin:/usr/bin:/bin", "LANG": "C.UTF-8", "LC_ALL": "C.UTF-8",
       "GIT_AUTHOR_NAME": "fx", "GIT_AUTHOR_EMAIL": "fx@fixture.invalid",
       "GIT_COMMITTER_NAME": "fx", "GIT_COMMITTER_EMAIL": "fx@fixture.invalid"}

def run_case(label, files, records, roots=()):
    if isinstance(records, str):
        records = [records]
    d = tempfile.mkdtemp(dir="/fast/scratch/hyaku/tameshi-r99/R99R1")
    def g(*a):
        return subprocess.run([GIT, "-C", d] + list(a), capture_output=True, env=ENV)
    assert g("init", "-q").returncode == 0
    top = g("rev-parse", "--show-toplevel").stdout.decode().strip()
    assert os.path.realpath(top) == os.path.realpath(d), "toplevel mismatch"
    for p, body in files.items():
        fp = os.path.join(d, p)
        os.makedirs(os.path.dirname(fp) or d, exist_ok=True)
        mode = "wb" if isinstance(body, bytes) else "w"
        with open(fp, mode) as fh:
            fh.write(body)
    assert g("add", "-A").returncode == 0
    assert g("commit", "-q", "-m", "fx").returncode == 0
    rev = g("rev-parse", "HEAD").stdout.decode().strip()
    cmd = ["python3", "-I", TOOL, "--repo", d, "--rev", rev]
    for r in records:
        cmd += ["--record", r]
    for r in roots:
        cmd += ["--receipt-root", r]
    p = subprocess.run(cmd, capture_output=True, text=True)
    summ = [l for l in p.stdout.splitlines() if l.startswith("TALLY_RECONCILE")]
    tall = [l for l in p.stdout.splitlines() if l.startswith("TALLY ")]
    print(f"### {label} rc={p.returncode}")
    for l in tall:
        print("   " + l)
    print("   SUMMARY: " + (" | ".join(summ) if summ else "(no summary)"))
    if p.stderr.strip():
        print("   STDERR: " + p.stderr.strip()[:200])
    if p.stdout.startswith("TALLY_SETUP") or p.stdout.startswith("TALLY_USAGE"):
        print("   SETUP/USAGE: " + p.stdout.strip()[:200])
    shutil.rmtree(d, ignore_errors=True)

REC = "p/round-3.md"
EV = ".ufo/evidence/round-3/a.out"

# resolution: tree path wins over round dirs (both exist, different content)
run_case("resolve-tree-wins",
         {REC: "selftest 2/2 (`a.out`)\n",
          "a.out": "C_SELFTEST: 2/2 pass\n",   # tree path, matches claim
          EV: "C_SELFTEST: 1/2 pass\n"},       # round dir, mismatches
         REC)
# reverse: tree path mismatches, round dir matches -> still tree path wins
run_case("resolve-tree-wins-mismatch",
         {REC: "selftest 2/2 (`a.out`)\n",
          "a.out": "C_SELFTEST: 1/2 pass\n",
          EV: "C_SELFTEST: 2/2 pass\n"},
         REC)
# round in directory, not basename -> no auto roots
run_case("resolve-round-in-dir-not-basename",
         {"p/round-3/notes.md": "selftest 2/2 (`a.out`)\n",
          EV: "C_SELFTEST: 2/2 pass\n"},
         "p/round-3/notes.md")
# uppercase ROUND in basename -> ROUND_RE is case-sensitive
run_case("resolve-round-uppercase",
         {"p/ROUND-3.md": "selftest 2/2 (`a.out`)\n",
          EV: "C_SELFTEST: 2/2 pass\n"},
         "p/ROUND-3.md")
# record name with round-NN mid-name: round-94-verifier-spotcheck.md
run_case("resolve-round-midname",
         {"x/round-94-verifier-spotcheck.md": "selftest 2/2 (`a.out`)\n",
          ".ufo/evidence/round-94/a.out": "C_SELFTEST: 2/2 pass\n"},
         "x/round-94-verifier-spotcheck.md")
# round-99 basename vs round-9 substring: record round-99.md must derive 99 not 9
run_case("resolve-round99-not-round9",
         {"x/round-99.md": "selftest 2/2 (`a.out`)\n",
          ".ufo/evidence/round-99/a.out": "C_SELFTEST: 2/2 pass\n",
          ".ufo/evidence/round-9/a.out": "C_SELFTEST: 1/2 pass\n"},
         "x/round-99.md")
# ambiguous between explicit root and auto root
run_case("resolve-ambiguous-explicit-vs-auto",
         {"p/notes.md": "selftest 2/2 (`a.out`)\n", "ev/a.out": "C_SELFTEST: 2/2 pass\n"},
         "p/notes.md", roots=["ev"])
run_case("resolve-explicit-root-ok",
         {"p/notes.md": "selftest 2/2 (`a.out`)\n", "ev/a.out": "C_SELFTEST: 2/2 pass\n"},
         "p/notes.md", roots=["ev"])
# ../ escape from round root into sibling round dir — must NOT resolve
run_case("resolve-dotdot-escape",
         {REC: "selftest 2/2 (`../round-4/a.out`)\n",
          ".ufo/evidence/round-4/a.out": "C_SELFTEST: 2/2 pass\n"}, REC)

# grading: token in comment line of receipt
run_case("grade-token-in-comment",
         {REC: "D_SELFTEST: 8/8 pass (`a.out`)\n",
          EV: "# D_SELFTEST: 9/9 pass\n"}, REC)
# N/M pass with trailing text
run_case("grade-trailing-text",
         {REC: "D_SELFTEST: 8/8 pass (`a.out`)\n",
          EV: "D_SELFTEST: 8/8 pass (details)\nD_SELFTEST: 8/8 pass\n"}, REC)
# binary / non-UTF-8 receipt
run_case("grade-binary-receipt",
         {REC: "selftest 2/2 (`a.out`)\n",
          EV: b"C_SELFTEST: 2/2 pass\n\xff\xfe\xfd binary \xd0\xb4\n"}, REC)
run_case("grade-binary-receipt-badtoken",
         {REC: "D_SELFTEST: 8/8 pass (`a.out`)\n",
          EV: b"D_SELFTEST: 8/8 pass\n\xff bad \xfe\xfd\n"}, REC)
# empty receipt
run_case("grade-empty-receipt-token",
         {REC: "D_SELFTEST: 8/8 pass (`a.out`)\n", EV: ""}, REC)
run_case("grade-empty-receipt-selftest",
         {REC: "selftest 2/2 (`a.out`)\n", EV: ""}, REC)
# lowercase token line in receipt vs token tally
run_case("grade-token-lowercase-in-receipt",
         {REC: "D_SELFTEST: 8/8 pass (`a.out`)\n", EV: "d_selftest: 8/8 pass\n"}, REC)
# selftest tally, receipt line lowercase *_selftest
run_case("grade-selftest-lowercase-in-receipt",
         {REC: "selftest 2/2 (`a.out`)\n", EV: "c_selftest: 2/2 pass\n"}, REC)
# receipt token line masked by replacement char
run_case("grade-token-broken-by-binary",
         {REC: "D_SELFTEST: 8/8 pass (`a.out`)\n", EV: b"D_SELFTEST: 8/8 pass\xff\n"}, REC)
# every-line rule: same token twice, one different
run_case("grade-every-line",
         {REC: "D_SELFTEST: 8/8 pass (`a.out`)\n",
          EV: "D_SELFTEST: 8/8 pass\nD_SELFTEST: 9/9 pass\n"}, REC)
print("DONE-batch3")
