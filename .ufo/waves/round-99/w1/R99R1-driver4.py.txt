#!/usr/bin/env python3
import os, subprocess, tempfile, shutil

LANE = "/home/vgpnk/Projects/shokunin/tameshi-shokunin"
TOOL = os.path.join(LANE, "tools/tally-reconcile.py")
GIT = "/usr/bin/git"
ENV = {"PATH": "/usr/local/bin:/usr/bin:/bin", "LANG": "C.UTF-8", "LC_ALL": "C.UTF-8",
       "GIT_AUTHOR_NAME": "fx", "GIT_AUTHOR_EMAIL": "fx@fixture.invalid",
       "GIT_COMMITTER_NAME": "fx", "GIT_COMMITTER_EMAIL": "fx@fixture.invalid"}

def run_case(label, files, records, roots=(), post=None):
    if isinstance(records, str):
        records = [records]
    d = tempfile.mkdtemp(dir="/fast/scratch/hyaku/tameshi-r99/R99R1")
    def g(*a):
        return subprocess.run([GIT, "-C", d] + list(a), capture_output=True, env=ENV)
    assert g("init", "-q").returncode == 0
    top = g("rev-parse", "--show-toplevel").stdout.decode().strip()
    assert os.path.realpath(top) == os.path.realpath(d), "toplevel mismatch"
    for p, body in files.items():
        fp = os.path.join(d, p)
        os.makedirs(os.path.dirname(fp) or d, exist_ok=True)
        mode = "wb" if isinstance(body, bytes) else "w"
        with open(fp, mode) as fh:
            fh.write(body)
    assert g("add", "-A").returncode == 0
    assert g("commit", "-q", "-m", "fx").returncode == 0
    rev = g("rev-parse", "HEAD").stdout.decode().strip()
    if post:
        post(d)
    cmd = ["python3", "-I", TOOL, "--repo", d, "--rev", rev]
    for r in records:
        cmd += ["--record", r]
    for r in roots:
        cmd += ["--receipt-root", r]
    p = subprocess.run(cmd, capture_output=True, text=True)
    summ = [l for l in p.stdout.splitlines() if l.startswith("TALLY_RECONCILE")]
    tall = [l for l in p.stdout.splitlines() if l.startswith("TALLY ")]
    print(f"### {label} rc={p.returncode}")
    for l in tall:
        print("   " + l)
    print("   SUMMARY: " + (" | ".join(summ) if summ else "(no summary)"))
    if p.stderr.strip():
        print("   STDERR: " + p.stderr.strip()[:200])
    if p.stdout.startswith("TALLY_SETUP") or p.stdout.startswith("TALLY_USAGE"):
        print("   SETUP/USAGE: " + p.stdout.strip()[:200])
    shutil.rmtree(d, ignore_errors=True)

REC = "p/round-3.md"
EV = ".ufo/evidence/round-3/a.out"
REC_BODY = "selftest 2/2 (`a.out`)\n"
EV_BODY = "C_SELFTEST: 2/2 pass\n"

run_case("tally-myselftest",
