#!/usr/bin/env python3
import os, subprocess, tempfile, shutil
LANE = "/home/vgpnk/Projects/shokunin/tameshi-shokunin"
TOOL = os.path.join(LANE, "tools/tally-reconcile.py")
GIT = "/usr/bin/git"
ENV = {"PATH": "/usr/local/bin:/usr/bin:/bin", "LANG": "C.UTF-8", "LC_ALL": "C.UTF-8",
       "GIT_AUTHOR_NAME": "fx", "GIT_AUTHOR_EMAIL": "fx@fixture.invalid",
       "GIT_COMMITTER_NAME": "fx", "GIT_COMMITTER_EMAIL": "fx@fixture.invalid"}
S = "/fast/scratch/hyaku/tameshi-r99/R99R1"
os.environ["TMPDIR"] = S

def run_case(label, files, records, roots=()):
    d = tempfile.mkdtemp(dir=S)
    def g(*a):
        return subprocess.run([GIT, "-C", d] + list(a), capture_output=True, env=ENV)
    assert g("init", "-q").returncode == 0
    top = g("rev-parse", "--show-toplevel").stdout.decode().strip()
    assert os.path.realpath(top) == os.path.realpath(d), "toplevel mismatch"
    for p, body in files.items():
        fp = os.path.join(d, p)
        os.makedirs(os.path.dirname(fp) or d, exist_ok=True)
        with open(fp, "wb" if isinstance(body, bytes) else "w") as fh:
            fh.write(body)
    assert g("add", "-A").returncode == 0
    assert g("commit", "-q", "-m", "fx").returncode == 0
    rev = g("rev-parse", "HEAD").stdout.decode().strip()
    cmd = ["python3", "-I", TOOL, "--repo", d, "--rev", rev]
    for r in records:
        cmd += ["--record", r]
    for r in roots:
        cmd += ["--receipt-root", r]
    p = subprocess.run(cmd, capture_output=True, text=True)
    print(f"### {label} rc={p.returncode}")
    for l in p.stdout.splitlines():
        print("   " + l)
    if p.stderr.strip():
        print("   STDERR: " + p.stderr.strip()[:200])
    shutil.rmtree(d, ignore_errors=True)

REC = "p/round-3.md"
EV = ".ufo/evidence/round-3/a.out"
# every-line: a token line whose suffix is not 'pass' is invisible to the grader
run_case("gap-every-line-fail-suffix",
         {REC: "D_SELFTEST: 8/8 pass (`a.out`)\n",
          EV: "D_SELFTEST: 8/8 pass\nD_SELFTEST: 9/9 fail\n"}, [REC])
run_case("gap-every-line-no-pass-suffix",
         {REC: "D_SELFTEST: 8/8 pass (`a.out`)\n",
          EV: "D_SELFTEST: 8/8 pass\nD_SELFTEST: 9/9\n"}, [REC])
run_case("gap-every-line-selftest-fail-suffix",
         {REC: "selftest 8/8 (`a.out`)\n",
          EV: "C_SELFTEST: 8/8 pass\nC_SELFTEST: 9/9 FAIL\n"}, [REC])
# citation grammar: trailing extension chain
run_case("cite-out-bak-chain",
         {REC: "selftest 2/2 (`a.out.bak`)\n", EV: "C_SELFTEST: 2/2 pass\n"}, [REC])
# scheme-less URL: docstring says 'not part of a URL'
run_case("cite-www-no-scheme-with-token-match",
         {REC: "selftest 2/2 see www.h.example/x/a.out\n",
          ".ufo/evidence/round-3/x/a.out": "C_SELFTEST: 2/2 pass\n"}, [REC])
# ambiguity: explicit root and auto round root both hold the basename
run_case("ambiguous-explicit-vs-auto-round",
         {REC: "selftest 2/2 (`a.out`)\n",
          "ev/a.out": "C_SELFTEST: 2/2 pass\n",
          EV: "C_SELFTEST: 2/2 pass\n"}, [REC], roots=["ev"])
# mixed-case token: Ab_SELFTEST is not a TOKEN
run_case("token-mixedcase-not-recognised",
         {REC: "Ab_SELFTEST: 2/2 pass (`a.out`)\n", EV: "Ab_SELFTEST: 2/2 pass\n"}, [REC])
# receipt line with token and 'pass' but digit-first token in receipt
run_case("token-in-receipt-digit-prefix",
         {REC: "D_SELFTEST: 8/8 pass (`a.out`)\n", EV: "1D_SELFTEST: 9/9 pass\nD_SELFTEST: 8/8 pass\n"}, [REC])
print("DONE-batch5")
