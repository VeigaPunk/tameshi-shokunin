## Verification script (reproduce: `python3 verify.py` in an empty scratch dir; reads git objects only)

Writes only to its scratch dir. Every line of output above comes from this script (STEP 2, 3, 4, 5 blocks) or from the
inline commands named in the Sweep and Provenance sections.

```python
import hashlib, importlib.util, json, subprocess, sys, os
LANE = '/home/vgpnk/Projects/shokunin/tameshi-shokunin'
PIN = '2f5512a8ab0771fb2eee463ec16db359174f5edd'
C0 = 'c0137451f5b7f22d65f748fc69f1b181f55e1378'
HEAD = '### Round-21 gate/probe registry — finite environment differential'
S = os.environ.get('S', '.')
ENV = {k: v for k, v in os.environ.items() if not k.startswith('GIT_')}
def git(*a):
    r = subprocess.run(['git', '-C', LANE] + list(a), capture_output=True, env=ENV)
    return r.returncode, r.stdout
def blob(rev, path):
    rc, out = git('cat-file', 'blob', f'{rev}:{path}')
    return out if rc == 0 else None
def mode(rev, path):
    rc, out = git('ls-tree', '-z', rev, '--', path)
    if rc: return None
    ents = [c for c in out.split(b'\x00') if c]
    return ents[0].split(b'\t',1)[0].split(b' ')[0].decode() if len(ents)==1 else None
def sha(b): return hashlib.sha256(b).hexdigest()

print('== STEP 2 ==')
print('PIN resolves:', git('rev-parse', '--verify', PIN+'^{commit}')[1].decode().strip())
print('C0 resolves:', git('rev-parse', '--verify', C0+'^{commit}')[1].decode().strip())
ledger = json.loads(blob(PIN, '.ufo/evidence/round-21/coldread/ledger.json'))
cands = [i for i,e in enumerate(ledger['entries']) if e.get('heading')==HEAD and e.get('read_rev')==C0 and any('rekey-r56-' in t for t in e.get('traces',[]))]
print('candidate indexes (heading+read_rev=C0+rekey-r56 trace):', cands)
assert len(cands)==1
idx = cands[0]; E = ledger['entries'][idx]
print('read_rev == C0:', E['read_rev']==C0, len(E['read_rev']))
dirs = {t.split('/traces/')[0] for t in E['traces']}
print('trace dirs:', dirs, 'all start rekey-r56:', all(t.startswith('.ufo/evidence/round-21/coldread/rekey-r56-') for t in E['traces']), 'n traces', len(E['traces']))
print('dir exists at PIN (mode):', [mode(PIN, d) for d in dirs])
print('wrong_expectations', E['wrong_expectations'], 'blocking_gaps', E['blocking_gaps'])
print('inputs count', len(E['inputs']))
print('LOG/SCORECARD in inputs:', [k for k in E['inputs'] if k.split('/')[-1] in ('LOG.md','SCORECARD.md')])
print('LOG/SCORECARD anywhere in inputs/external:', [k for k in list(E['inputs'])+list(E.get('external_inputs',{})) if 'LOG.md' in k or 'SCORECARD.md' in k])

print('== STEP 3 ==')
src = blob(PIN, '.ufo/evidence/round-18/measure.py')
open(f'{S}/r18-measure-pin.py','wb').write(src)
spec = importlib.util.spec_from_file_location('r18m', f'{S}/r18-measure-pin.py')
m = importlib.util.module_from_spec(spec); spec.loader.exec_module(m)
print('sha of frozen r18 measure.py at PIN:', sha(src))
secs = m.sectionize(blob(C0, 'SKILL.md'))
print('sections at C0:', len(secs))
match = [s for s in secs if s['heading']==HEAD]
print('sections with heading at C0:', [(s['start'],s['end'],s['sha256']) for s in match])
sec = [s for s in match if s['start']==E['start']]
print('section sha == entry sha:', len(sec)==1 and sec[0]['sha256']==E['sha256'], '| end match:', len(sec)==1 and sec[0]['end']==E['end'])
for p, v in sorted(E['inputs'].items()):
    b = blob(PIN, p); got = sha(b) if b is not None else None
    print(f'  {"OK " if got==v else "BAD"} {p} mode={mode(PIN,p)} entry={v[:12]} pin={(got or "MISSING")[:12]}')
for k, v in sorted(E.get('external_inputs',{}).items()):
    rev, path = k.split('@',1)[1].split(':',1)
    b = blob(C0, path); got = sha(b) if b is not None else None
    print(f'  {"OK " if got==v else "BAD"} external C0 {path}')

print('== STEP 4 ==')
r = m.entry_reason(LANE, PIN, E, True)
print('entry_reason(lane, PIN, entry, True) =', repr(r))
print('entry_credit_reason =', repr(m.entry_credit_reason(E)), '| input_reason =', repr(m.input_reason(LANE, PIN, E)), '| trace_reason =', repr(m.trace_reason(LANE, PIN, E, True)))

print('== STEP 5 ==')
md = blob(PIN, '.ufo/waves/round-56/w2/T56W2Read15.md')
print('md bytes', len(md), 'sha', sha(md))
rep = json.loads(blob(PIN, '.ufo/waves/round-56/w2/T56W2Read15.json'))
print('heading match:', rep['heading']==E['heading'], '| cand==C0:', rep['cand']==C0, '| section_sha256==sha:', rep['section_sha256']==E['sha256'])
```
