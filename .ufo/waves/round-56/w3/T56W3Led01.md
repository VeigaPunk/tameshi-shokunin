```python
import importlib.util, hashlib, json, subprocess, sys
R = '/home/vgpnk/Projects/shokunin/tameshi-shokunin'
PIN = '2f5512a8ab0771fb2eee463ec16db359174f5edd'

H = '## Context gate classification and explicit exit classes \u2014 round 15'
S = '/fast/scratch/hyaku/tameshi-r56/T56W3Led01'

def gshow(rev, path):
    r = subprocess.run(['git', '-C', R, 'show', '%s:%s' % (rev, path)], capture_output=True)
    return r.returncode, r.stdout
def sha(b): return hashlib.sha256(b).hexdigest()
def mode(rev, path):
    r = subprocess.run(['git', '-C', R, 'ls-tree', '-z', rev, '--', path], capture_output=True)
    return r.stdout.split(b'\t')[0].split(b' ')[0].decode() if r.stdout else None

rc, mod_bytes = gshow(PIN, '.ufo/evidence/round-18/measure.py'); assert rc == 0
open(S + '/measure-at-PIN.py', 'wb').write(mod_bytes)
spec = importlib.util.spec_from_file_location('r18measure', S + '/measure-at-PIN.py')
m = importlib.util.module_from_spec(spec); spec.loader.exec_module(m)

rc, led = gshow(PIN, '.ufo/evidence/round-21/coldread/ledger.json')
d = json.loads(led)
idx = [i for i, e in enumerate(d['entries']) if e.get('heading') == H and any('rekey-r56-' in t for t in e.get('traces', []))]
assert len(idx) == 1
e = d['entries'][idx[0]]
print('entry index', idx[0], 'read_rev == C0:', e['read_rev'] == C0)
print('trace dirs:', sorted({'/'.join(t.split('/')[:5]) for t in e['traces']}))
print('wrong_expectations', e['wrong_expectations'], 'blocking_gaps', e['blocking_gaps'])

rc, skill = gshow(C0, 'SKILL.md'); assert rc == 0
secs = m.sectionize(skill)
match = [s for s in secs if s['heading'] == H and s['start'] == e['start'] and s['end'] == e['end']]
print('SECTION SHA MATCH:', bool(match) and match[0]['sha256'] == e['sha256'])

bad = []
for k, v in sorted(e['inputs'].items()):
    rc, data = gshow(PIN, k)
    if rc != 0 or sha(data) != v: bad.append(k)
print('inputs at PIN, mismatches:', bad, 'of', len(e['inputs']))
print('entry_reason(R, PIN, entry, True) =', repr(m.entry_reason(R, PIN, e, True)))

rep = json.loads(gshow(PIN, '.ufo/waves/round-56/w2/T56W2Read01.json')[1])
print('reader heading ok:', rep['heading'] == H, 'cand ok:', rep['cand'] == C0,
      'section_sha ok:', rep['section_sha256'] == e['sha256'])
```
