import os, sys, subprocess, runpy, json, hashlib
from pathlib import Path

os.environ['TMPDIR'] = str(root / 'tmp')
os.environ['PYTHONDONTWRITEBYTECODE'] = '1'
original = subprocess.run
checks = 0
def guarded(args, *a, **kw):
    global checks
    if isinstance(args, (list, tuple)) and args and args[0] == 'git':
        if '-C' not in args:
            raise RuntimeError('fixture git command missing explicit -C: %r' % args)
        repo = Path(args[args.index('-C') + 1]).resolve()
        if not repo.is_relative_to(root):
            raise RuntimeError('fixture outside scratch: %s' % repo)
        if any(command in args for command in ('add', 'commit', 'config', 'checkout', 'reset', 'update-index')):
            top = original(['git', '-C', str(repo), 'rev-parse', '--show-toplevel'], capture_output=True, check=True).stdout.decode().strip()
            assert top == str(repo), (top, repo)
            checks += 1
    return original(args, *a, **kw)
subprocess.run = guarded
script = sys.argv[1]
sys.argv = [script, '--selftest']
try:
    runpy.run_path(script, run_name='__main__')
finally:
    print('FIXTURE_TOPLEVEL_ASSERTIONS=%d' % checks, file=sys.stderr)
